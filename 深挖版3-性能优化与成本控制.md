### 1.3 Token 成本监控器（必须做）

```java
package com.example.agentcost;

import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;

/**
 * Token 成本监控器 —— 每次 LLM 调用都记账
 */
@Component
public class TokenCostMonitor {

    private final MeterRegistry registry;
    private final ConcurrentHashMap<String, ModelPricing> pricingMap = new ConcurrentHashMap<>();

    public TokenCostMonitor(MeterRegistry registry) {
        this.registry = registry;
        initPricing();
    }

    /**
     * 2026 年 6 月实测定价
     */
    private void initPricing() {
        // 单位：$ / 1M tokens
        registerModel("deepseek-v3.2", 0.14, 0.28, 0.0028);
        registerModel("deepseek-v4-flash", 0.14, 0.28, 0.0028);
        registerModel("gpt-5.4", 2.50, 10.00, 0.50);
        registerModel("gpt-5.5", 5.00, 20.00, 0.50);
        registerModel("claude-opus-4.6", 5.00, 25.00, 0.50);
        registerModel("claude-sonnet-4.6", 3.00, 15.00, 0.30);
        registerModel("gemini-2.5-pro", 1.25, 5.00, 0.31);
        registerModel("gemini-2.5-flash", 0.30, 1.20, 0.075);
        registerModel("qwen3-max", 0.40, 1.20, 0.04);
        registerModel("glm-4.6", 0.50, 1.50, 0.05);
    }

    private void registerModel(String name, double input, double output, double cacheRead) {
        pricingMap.put(name, new ModelPricing(input, output, cacheRead));
    }

    /**
     * 记录一次 LLM 调用
     */
    public void record(String model, String tenantId, String userId,
                       long promptTokens, long completionTokens,
                       long cacheReadTokens, long cacheWriteTokens) {
        ModelPricing p = pricingMap.get(model);
        if (p == null) {
            registry.counter("llm.unknown_model", "model", model).increment();
            return;
        }

        // 计算费用（美元）
        double cost = (cacheReadTokens * p.cacheRead
                    + cacheWriteTokens * p.input   // 缓存写也按 input 计（部分厂商 1.25x）
                    + (promptTokens - cacheReadTokens - cacheWriteTokens) * p.input
                    + completionTokens * p.output) / 1_000_000.0;

        // Micrometer 指标（接 Prometheus / Grafana）
        registry.counter("llm.tokens", "model", model, "type", "input").increment(promptTokens);
        registry.counter("llm.tokens", "model", model, "type", "output").increment(completionTokens);
        registry.counter("llm.tokens", "model", model, "type", "cache_read").increment(cacheReadTokens);
        registry.counter("llm.cost.usd", "model", model, "tenant", tenantId).increment(cost);

        // 内存累计（按租户/用户查账单）
        String tenantKey = "cost:" + LocalDate.now() + ":" + tenantId;
        registry.counter("llm.cost.by_tenant", "tenant", tenantId).increment(cost);

        // 异常大请求监控
        if (promptTokens > 50000) {
            registry.counter("llm.large_request", "model", model, "tenant", tenantId).increment();
        }
    }

    /**
     * 查询今日某租户的成本
     */
    public BigDecimal getTodayCost(String tenantId) {
        // 实际从 Prometheus 查 sum(llm_cost_usd{tenant=xxx})
        return BigDecimal.ZERO;  // 简化
    }

    private record ModelPricing(double input, double output, double cacheRead) {}
}
```

## 第二部分：缓存体系（降本的最大杠杆）

**2026 年的数据：缓存读比 cache miss 便宜 5-50 倍。**

**把缓存命中率从 0% 提到 70%，账单直接降 50% 以上。**

### 2.1 缓存的 4 层架构

```
┌─────────────────────────────────────────────────────────┐
│ L1 客户端缓存（HTTP 304 / ETag）                          │
│    ↓ miss                                                 │
│ L2 进程内缓存（Caffeine，~1ms，TTL 5min）                  │
│    ↓ miss                                                 │
│ L3 分布式缓存（Redis，~5ms，TTL 1h）                       │
│    ↓ miss                                                 │
│ L4 厂商 Prompt Cache（DeepSeek/Claude/Gemini，~200ms）    │
│    ↓ miss                                                 │
│ 真实 LLM API（~1-3s）                                     │
└─────────────────────────────────────────────────────────┘
```

### 2.2 L1+L2: Caffeine 客户端缓存

```xml
<dependency>
    <groupId>com.github.ben-manes.caffeine</groupId>
    <artifactId>caffeine</artifactId>
</dependency>
```

```java
@Configuration
public class CacheConfig {

    /**
     * L1+L2 缓存：进程内 + TTL
     * 适合"短时间重复问题"的场景（比如用户连续点 3 次"重试"按钮）
     */
    @Bean
    public Cache<String, String> promptCache() {
        return Caffeine.newBuilder()
            .maximumSize(10_000)              // 最多 1 万条
            .expireAfterWrite(Duration.ofMinutes(5))   // 5 分钟过期
            .recordStats()                    // 开启统计
            .build();
    }
}
```

**使用**：

```java
@Service
public class CachedLlmService {

    @Autowired private ChatClient chatClient;
    @Autowired private Cache<String, String> promptCache;
    @Autowired private TokenCostMonitor monitor;

    public String chat(String prompt, String model) {
        // 1. 生成缓存 key（用 prompt 的 hash）
        String key = model + ":" + DigestUtils.sha256Hex(prompt);

        // 2. 查缓存
        String cached = promptCache.getIfPresent(key);
        if (cached != null) {
            monitor.recordCacheHit(model, "l1");
            return cached;
        }

        // 3. 缓存未命中，调 LLM
        String response = chatClient.prompt()
            .user(prompt)
            .call()
            .content();

        // 4. 写回缓存
        promptCache.put(key, response);
        return response;
    }
}
```

**⚠️ 缓存的 4 个大坑**：

| 坑 | 现象 | 解法 |
|---|---|---|
| **缓存了过期内容** | 改了 Prompt 但还返回旧答案 | 用 Prompt 版本号做 key 前缀 |
| **缓存了敏感数据** | 不同用户数据混了 | key 包含 `userId + tenantId` |
| **缓存爆炸** | 内存 OOM | 设 `maximumSize` + LRU 淘汰 |
| **没统计命中率** | 不知道缓存有没有用 | 一定要 `recordStats()` + 接 Micrometer |

### 2.3 L3: Redis 分布式缓存（多实例共享）

```java
@Service
public class RedisLlmCache {

    @Autowired private RedisTemplate<String, String> redis;
    @Autowired private ObjectMapper objectMapper;

    /**
     * 用 Redis Hash 存（key: tenant_id，field: prompt_hash，value: 完整 response + meta）
     */
    public Optional<CachedLlmResponse> get(String tenantId, String promptHash) {
        String key = "llm:cache:" + tenantId;
        String json = (String) redis.opsForHash().get(key, promptHash);
        if (json == null) return Optional.empty();

        try {
            CachedLlmResponse cached = objectMapper.readValue(json, CachedLlmResponse.class);
            // 检查是否过期
            if (cached.expiresAt < System.currentTimeMillis()) {
                redis.opsForHash().delete(key, promptHash);
                return Optional.empty();
            }
            return Optional.of(cached);
        } catch (Exception e) {
            return Optional.empty();
        }
    }

    public void put(String tenantId, String promptHash, String response, long ttlSeconds) {
        CachedLlmResponse cached = new CachedLlmResponse(
            response,
            System.currentTimeMillis() + ttlSeconds * 1000
        );
        try {
            redis.opsForHash().put(
                "llm:cache:" + tenantId,
                promptHash,
                objectMapper.writeValueAsString(cached)
            );
            // 整个 hash 24h 过期（防止长期无访问堆积）
            redis.expire("llm:cache:" + tenantId, Duration.ofHours(24));
        } catch (Exception e) {
            // 写失败不影响主流程
        }
    }

    /**
     * 模糊查询（语义缓存的预备步骤）
     */
    public List<String> findSimilar(String tenantId, String promptPrefix) {
        return redis.opsForHash().keys("llm:cache:" + tenantId).stream()
            .map(Object::toString)
            .filter(k -> k.startsWith(promptPrefix))
            .collect(Collectors.toList());
    }
}

record CachedLlmResponse(String content, long expiresAt) {}

### 2.4 L4: 厂商原生 Prompt Cache（最猛）

**这是 2025-2026 年才普及的"硬核省钱能力"。**

**原理**：把长 System Prompt 在厂商服务器上缓存 5-60 分钟，重复请求相同前缀直接命中缓存。

#### DeepSeek Prompt Cache 用法

```java
/**
 * DeepSeek 自动启用 Prompt Cache（无需特殊 API）
 * 命中条件：相同前缀的 tokens ≥ 1024
 * 缓存时长：默认 5-10 分钟
 */
@Service
public class DeepSeekCachedChat {

    @Autowired private ChatClient chatClient;

    /**
     * 把不变化的内容（System Prompt + 知识库内容）放在最前面
     * 把每次都变化的内容（用户问题）放在最后
     * —— 这就是 Prompt Cache 的核心技巧
     */
    public String askWithRag(String question, List<Document> context) {
        return chatClient.prompt()
            // ↓↓↓ 静态部分（命中缓存的关键）
            .system("""
                你是企业知识助手，严格根据 context 回答。
                """)
            .user(u -> u.text("""
                参考以下文档回答问题：
                
                {context}
                
                ---
                
                问题：{question}
                """)
                .param("context", formatContext(context))   // 长文本放中间
                .param("question", question)                 // 动态放最后
            )
            .options(DeepSeekChatOptions.builder()
                .model("deepseek-chat")
                .build())
            .call()
            .content();
    }
}
```

**结构原则**：
```
[系统提示 - 不变] → [RAG 文档 - 大部分时间不变] → [历史消息 - 累积] → [用户问题 - 每次变]
↑ 缓存命中区                          ↑              ↑
```

### 2.5 缓存命中率的"监控 + 优化"循环

```java
@Component
public class CacheMetricsAdvisor implements BaseAdvisor {

    private final Counter l1Hit = Counter.builder("llm.cache").tag("level", "l1").register(registry);
    private final Counter l2Hit = Counter.builder("llm.cache").tag("level", "l2").register(registry);
    private final Counter l3Hit = Counter.builder("llm.cache").tag("level", "l3").register(registry);
    private final Counter miss = Counter.builder("llm.cache").tag("level", "miss").register(registry);

    @Override
    public ChatClientRequest before(ChatClientRequest request, AdvisorChain chain) {
        request.context().put("cache_check_time", System.currentTimeMillis());
        return request;
    }

    @Override
    public ChatClientResponse after(ChatClientResponse response, AdvisorChain chain) {
        Long start = (Long) response.context().get("cache_check_time");
        long latency = System.currentTimeMillis() - (start != null ? start : 0);

        // 根据 latency 反推命中哪一层（命中率分析）
        if (latency < 5) l1Hit.increment();
        else if (latency < 50) l2Hit.increment();
        else if (latency < 500) l3Hit.increment();
        else miss.increment();

        return response;
    }
}
```

**每周跑一次分析**：
- L1 命中率 < 30% → Caffeine TTL 太短
- L3 命中率 < 50% → Redis key 设计有问题
- 总命中率 < 60% → 考虑优化 Prompt 结构

### 3.1 路由策略矩阵

### 3.2 路由器实现

```java
package com.example.agentcost;

import org.springframework.stereotype.Service;

import java.util.*;
import java.util.regex.Pattern;

/**
 * 智能模型路由器
 */
@Service
public class ModelRouter {

    /**
     * 路由决策
     * @return 选中的模型名
     */
    public RouteDecision route(String userMessage, String systemPrompt, UserContext ctx) {
        // 1. 检测问题特征
        QuestionFeatures features = analyze(userMessage, systemPrompt);

        // 2. 按规则路由
        String model = pickModel(features);

        // 3. 返回决策（包含原因，方便审计）
        return new RouteDecision(model, features.reason(), features.estimatedTokens());
    }

    private QuestionFeatures analyze(String msg, String sys) {
        int length = msg.length();
        boolean hasCode = msg.contains("```") || msg.contains("public class") || msg.contains("function ");
        boolean hasChinese = msg.codePoints().anyMatch(c -> c >= 0x4E00 && c <= 0x9FFF);
        boolean hasImage = msg.contains("![");  // Markdown 图片
        boolean hasComplexKeyword = Pattern.compile("架构|设计模式|算法|分布式|并发|优化|debug")
            .matcher(msg).find();
        boolean isGreeting = Pattern.compile("^(你好|hi|hello|嗨|hey|在吗)\\W*$", Pattern.CASE_INSENSITIVE)
            .matcher(msg.trim()).find();

        StringBuilder reason = new StringBuilder();
        if (isGreeting) reason.append("闲聊 → ");
        else if (hasCode && hasComplexKeyword) reason.append("复杂代码 → ");
        else if (hasCode) reason.append("代码问题 → ");
        else if (hasComplexKeyword) reason.append("深度推理 → ");
        else if (hasImage) reason.append("多模态 → ");
        else if (hasChinese) reason.append("中文问题 → ");
        else reason.append("一般问题 → ");

        return new QuestionFeatures(length, hasCode, hasChinese, hasImage, hasComplexKeyword, isGreeting, reason.toString());
    }

    private String pickModel(QuestionFeatures f) {
        // 优先级：特殊 > 复杂 > 中等 > 简单
        if (f.hasImage) return "gpt-5.4";                      // 多模态必须 GPT
        if (f.isGreeting) return "deepseek-v3.2";              // 闲聊用最便宜
        if (f.hasCode && f.hasComplexKeyword) return "claude-opus-4.6";  // 复杂代码用 Opus
        if (f.hasComplexKeyword) return "claude-sonnet-4.6";   // 深度推理用 Sonnet
        if (f.hasChinese) return "qwen3-max";                  // 中文 + 国产合规
        if (f.length < 50) return "deepseek-v3.2";             // 短问题用便宜
        return "claude-sonnet-4.6";                            // 默认 Sonnet（平衡）
    }
}

record QuestionFeatures(int length, boolean hasCode, boolean hasChinese,
                       boolean hasImage, boolean hasComplexKeyword,
                       boolean isGreeting, String reason) {
    long estimatedTokens() { return length / 2; }  // 粗略估计
}

record RouteDecision(String model, String reason, long estimatedTokens) {}
```

### 3.4 动态降级（容灾必备）

**核心**：当旗舰模型挂了，自动降级到便宜模型。

```java
@Service
public class FallbackChatService {

    @Autowired private Map<String, ChatClient> clientMap;
    @Autowired private CircuitBreakerRegistry circuitBreakerRegistry;

    /**
     * 三级降级：旗舰 → 中档 → 兜底
     */
    public String chatWithFallback(String userMessage) {
        List<String> chain = List.of("claude-opus-4.6", "claude-sonnet-4.6", "deepseek-v3.2");

        for (String model : chain) {
            try {
                CircuitBreaker cb = circuitBreakerRegistry.circuitBreaker(model);
                return cb.executeSupplier(() ->
                    clientMap.get(model).prompt()
                        .user(userMessage)
                        .call()
                        .content()
                );
            } catch (CallNotPermittedException e) {
                // 熔断器开着，跳过这个模型
                log.warn("[Fallback] {} circuit open, trying next", model);
            } catch (Exception e) {
                log.error("[Fallback] {} failed, trying next", model, e);
            }
        }
        throw new RuntimeException("所有模型都挂了，请稍后重试");
    }
}
```

## 第四部分：JVM + 架构调优（Java 工程师的老本行）

**这部分是咱 Java 工程师的"舒适区"，但 Agent 时代有些新特性要特别关注。**

### 4.1 LLM 应用的 GC 调优

**LLM 应用的特殊之处**：
- **流式输出时会产生大量小对象**（SSE chunks）
- **Prompt 字符串可能 100KB+**（大对象直接进老年代）
- **Vector Embedding（1024 维 float[]）** 在堆里反复创建

**调优方案**：

```bash
# G1GC 配置（Java 17+，默认就用 G1）
-XX:+UseG1GC
-XX:MaxGCPauseMillis=100           # Agent 场景延迟敏感
-XX:G1HeapRegionSize=16m           # 配合大对象
-XX:InitiatingHeapOccupancyPercent=45  # 提早启动并发周期
-XX:+G1UseStringDeduplication      # 字符串去重，Prompt 重复部分多

# 大对象阈值（防止大 Prompt 反复 GC）
-XX:PretenureSizeThreshold=1m      # > 1MB 直接老年代

# 内存
-Xms4g -Xmx4g                      # Agent 进程固定堆大小（避免动态伸缩）
-XX:MetaspaceSize=512m
```

**辅助：用 Netty 的 PooledByteBufAllocator**（WebFlux 默认）减少堆外内存拷贝。

### 4.3 向量计算的 SIMD 加速

**Embedding 检索是热点，Java 21 的 Vector API 能提速 5-10x。**

```java
/**
 * 用 Java 21 Vector API 计算向量相似度
 */
public class VectorSimilarity {

    /**
     * 普通版：~50ms（1024 维）
     */
    public static float cosineSimilarity(float[] a, float[] b) {
        float dot = 0, normA = 0, normB = 0;
        for (int i = 0; i < a.length; i++) {
            dot += a[i] * b[i];
            normA += a[i] * a[i];
            normB += b[i] * b[i];
        }
        return dot / (float) (Math.sqrt(normA) * Math.sqrt(normB));
    }

    /**
     * SIMD 版：~5-10ms（1024 维）
     * 需 Java 21+ 开启 --add-modules jdk.incubator.vector
     */
    public static float cosineSimilaritySimd(float[] a, float[] b) {
        int length = a.length;
        var species = FloatVector.SPECIES_PREFERRED;
        int upperBound = species.loopBound(length);

        var dotSum = FloatVector.zero(species);
        var normASum = FloatVector.zero(species);
        var normBSum = FloatVector.zero(species);

        for (int i = 0; i < upperBound; i += species.length()) {
            var va = FloatVector.fromArray(species, a, i);
            var vb = FloatVector.fromArray(species, b, i);
            dotSum = va.fma(vb, dotSum);
            normASum = va.fma(va, normASum);
            normBSum = vb.fma(vb, normBSum);
        }

        float dot = dotSum.reduceLanes(VectorOperators.ADD);
        float normA = (float) Math.sqrt(normASum.reduceLanes(VectorOperators.ADD));
        float normB = (float) Math.sqrt(normBSum.reduceLanes(VectorOperators.ADD));

        return dot / (normA * normB);
    }
}
```

### 4.5 限流（防 LLM 被刷爆）

```java
@Configuration
public class RateLimitConfig {

    /**
     * 全局限流：每秒 1000 次 LLM 调用
     */
    @Bean
    public RateLimiter globalRateLimiter() {
        return RateLimiter.create(1000.0);
    }
}

@Component
public class GlobalRateLimitAdvisor implements BaseAdvisor {

    @Autowired private RateLimiter globalRateLimiter;
    @Autowired private RedisTemplate<String, Long> redis;

    @Override
    public ChatClientRequest before(ChatClientRequest request, AdvisorChain chain) {
        // 1. 全局限流（保护 LLM 厂商调用额度）
        if (!globalRateLimiter.tryAcquire(100, TimeUnit.MILLISECONDS)) {
            throw new RateLimitExceededException("系统繁忙，请稍后重试");
        }

        // 2. 每用户限流
        String userId = UserContext.get().userId();
        String key = "ratelimit:" + userId + ":" + (System.currentTimeMillis() / 60000);
        Long count = redis.opsForValue().increment(key, 1);
        redis.expire(key, 65, TimeUnit.SECONDS);
        if (count != null && count > 20) {
            throw new RateLimitExceededException("您调用过于频繁，请 1 分钟后再试");
        }

        // 3. 每租户限流（保护账单）
        String tenantId = UserContext.get().tenantId();
        String tenantKey = "ratelimit:tenant:" + tenantId + ":" + (System.currentTimeMillis() / 60000);
        Long tenantCount = redis.opsForValue().increment(tenantKey, 1);
        if (tenantCount != null && tenantCount > 1000) {
            throw new RateLimitExceededException("租户调用额度已用完");
        }

        return request;
    }
}
